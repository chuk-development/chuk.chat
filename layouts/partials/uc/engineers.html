class AuthService {
  AuthService(this._store);
  final TokenStore _store;
 
  Future<Session> refresh() async {
    final token = await _store.read();
    if (token.isExpired) {
      return _renew(token);
    }
    return Session(token);
  }
 
  Future<Session> _renew(Token old) async {
    final fresh = await _api.refresh(old);
    await _store.write(fresh);
    return Session(fresh);
  }
}
